# Web dashboard backend (server/main.py — FastAPI JSON API for the Next.js UI;
# wraps airtable_client + finance, no new business logic). Run with:
#   uvicorn server.main:app --port 8000
# FastAPI + Starlette are pinned TOGETHER, inside the ONE window that satisfies
# every user of Starlette in this shared requirements file:
#   * streamlit 1.58.0 (app.py) requires  starlette>=0.40.0
#   * fastapi 0.115.6   (server/)   requires  starlette>=0.40.0,<0.42.0
#   => the overlap is [0.40.0, 0.42.0); starlette 0.41.3 sits inside it.
# Staying <0.42.0 (and fastapi <0.116) also keeps the build clear of the
# `DEFAULT_EXCLUDED_CONTENT_TYPES` import (added in Starlette 0.42.0 / FastAPI
# 0.116) that crashed earlier deploys. Do NOT downgrade fastapi below 0.115 —
# older FastAPI caps starlette <0.40 and hard-conflicts with streamlit.
fastapi==0.115.6         # REST endpoints for the Next.js dashboard
starlette==0.41.3        # ASGI toolkit shared by FastAPI AND streamlit 1.58
uvicorn[standard]==0.34.0  # ASGI server (dev + prod)
PyJWT==2.10.1            # JWT auth (auth.py) — Bearer tokens for the dashboard

# --- Transitive binary deps, pinned to segfault-proof the deploy ------------
# Exit 139 (SIGSEGV) on Render comes from compiled wheels mismatching the
# interpreter build, so the packages that ship native code are locked to
# versions with mature manylinux cp311 wheels (Python is pinned to 3.11 in
# the Dockerfile — the effective pin, since render.yaml uses runtime: docker
# — and in runtime.txt for any non-Docker deploy).
pydantic==2.11.10        # pulled in by openai; pydantic-core is its Rust half
pydantic-core==2.33.2    # the exact core build pydantic 2.11.10 requires
pyarrow==20.0.0          # pulled in by streamlit (dataframe serialization)
