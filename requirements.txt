# Web dashboard backend (server/main.py — FastAPI JSON API for the Next.js UI;
# wraps airtable_client + finance, no new business logic). Run with:
#   uvicorn server.main:app --port 8000
# FastAPI + Starlette are pinned TOGETHER to a known-good, mutually compatible
# pair. FastAPI 0.109.2 predates the `DEFAULT_EXCLUDED_CONTENT_TYPES` import
# (introduced in FastAPI 0.116 / Starlette 0.42.0), so this pair is IMMUNE to:
#   ImportError: cannot import name 'DEFAULT_EXCLUDED_CONTENT_TYPES'
#                from 'starlette.middleware.gzip'
# — regardless of what a build's resolver would otherwise pick. FastAPI 0.109.2
# requires starlette>=0.35,<0.37, so 0.36.3 is the exact matching build. Do NOT
# bump one without the other. Verified locally: `uvicorn server.main:app` starts
# and serves (health/meta/openapi/auth), with pydantic 2.11.10, against live
# Airtable.
fastapi==0.109.2         # REST endpoints for the Next.js dashboard
starlette==0.36.3        # ASGI toolkit under FastAPI — kept in lockstep
uvicorn[standard]==0.34.0  # ASGI server (dev + prod)
PyJWT==2.10.1            # JWT auth (auth.py) — Bearer tokens for the dashboard

# --- Transitive binary deps, pinned to segfault-proof the deploy ------------
# Exit 139 (SIGSEGV) on Render comes from compiled wheels mismatching the
# interpreter build, so the packages that ship native code are locked to
# versions with mature manylinux cp311 wheels (Python is pinned to 3.11 in
# the Dockerfile — the effective pin, since render.yaml uses runtime: docker
# — and in runtime.txt for any non-Docker deploy).
pydantic==2.11.10        # pulled in by openai; pydantic-core is its Rust half
