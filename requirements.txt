# ==========================================================================
# Web backend dependencies — FastAPI + Uvicorn ONLY.
# ==========================================================================
# The product has migrated to Next.js (web/) + FastAPI (server/). This file is
# scoped to EXACTLY what the FastAPI backend imports, so the API image stays
# small.
#
#   Deploy:  uvicorn server.main:app --host 0.0.0.0 --port $PORT
#
# The Streamlit app and its Flask webhook were deleted outright once the
# migration completed, so there is no longer a legacy requirements file to keep
# in step — this is the only dependency list in the repo.
# --------------------------------------------------------------------------

fastapi==0.115.6           # REST API (server/main.py)
# Starlette is pinned in lockstep with FastAPI. 0.41.3 is inside FastAPI
# 0.115.6's required range (>=0.40,<0.42) and stays below 0.42.0, so the build
# never imports DEFAULT_EXCLUDED_CONTENT_TYPES (added in Starlette 0.42.0 /
# FastAPI 0.116). Do not bump one without the other.
starlette==0.41.3
uvicorn[standard]==0.34.0  # ASGI server (dev + prod)

PyJWT==2.10.1              # JWT bearer auth (auth.py — `import jwt`)
stripe==15.4.0             # billing webhook signature check (server/webhooks.py)
python-dotenv==1.1.1       # .env loading (config.py)
email-validator==2.3.0     # pydantic EmailStr on the registration endpoint
python-multipart==0.0.32   # REQUIRED by FastAPI for UploadFile — without it the
                           # invoice-scan endpoint raises at import time, not on
                           # first request

# --- Invoice OCR ----------------------------------------------------------

# --- PostgreSQL (Supabase) ------------------------------------------------
